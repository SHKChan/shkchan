# HOW TO PUBLISH A POST
# 1. Copy this file into this folder and rename it, e.g. my-first-post.md.
#    The file name becomes the URL: /blog/posts/my-first-post
#    (files starting with "_" like this one are never published).
# 2. Fill in the fields below, write the article under the second ---, then push.
#    The home page, category pages and tag pages update automatically.

title: "Your Post Title"                       # required
description: "One sentence on what this post is about."  # required; shown on cards and in search results
date: 2026-10-26                               # required; YYYY-MM-DD, posts are sorted newest first
category: "Gaming"                             # required; one category per post, a new name creates a new category page
tags: ["C++", "Game Mechanics"]                # optional keywords; each one gets its own tag page
# updated: 2026-11-01                          # optional; shows "Updated" on the post
# draft: true                                  # optional; true hides the post from the live site
---
